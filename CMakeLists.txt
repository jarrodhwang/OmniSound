cmake_minimum_required(VERSION 3.25)
project(OmniSound VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
add_library(omni_core src/core/model.cpp src/core/dsp.cpp src/core/signal_quality.cpp src/core/routing.cpp src/core/localization.cpp)
target_include_directories(omni_core PUBLIC include third_party)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
  target_compile_options(omni_core PRIVATE /W4 /permissive- /utf-8 /EHsc)
endif()
if(WIN32)
  add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_definitions(omni_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_library(omni_win src/win/platform.cpp src/win/audio.cpp src/win/ipc.cpp)
  target_link_libraries(omni_win PUBLIC omni_core ole32 uuid avrt propsys advapi32 shell32 cfgmgr32 swdevice)
  add_executable(OmniSoundEngine src/engine/main.cpp src/engine/router.cpp src/engine/calibration.cpp src/win/version.rc)
  target_link_libraries(OmniSoundEngine PRIVATE omni_win)
  target_link_options(OmniSoundEngine PRIVATE /SUBSYSTEM:WINDOWS /ENTRY:wmainCRTStartup)
  add_executable(OmniSoundService src/service/main.cpp src/service/devices.cpp src/win/version.rc)
  target_link_libraries(OmniSoundService PRIVATE omni_win setupapi wtsapi32)
  # The product UI is a native WinUI 3 / C++/WinRT MSBuild project. CMake
  # continues to own shared engine/service code and the platform-independent tests.
  add_custom_target(OmniSound SOURCES
    src/ui/winui/OmniSound.vcxproj
    src/ui/winui/Project.idl src/ui/winui/MainWindow.h src/ui/winui/pch.cpp
    src/ui/winui/App.xaml src/ui/winui/App.xaml.cpp src/ui/winui/App.xaml.h
    src/ui/winui/MainWindow.xaml src/ui/winui/MainWindow.xaml.cpp src/ui/winui/MainWindow.xaml.h
    src/ui/winui/OmniSound.manifest
    src/ui/controller.hpp src/ui/controller.cpp)
  add_executable(OmniSoundProbe src/tools/probe.cpp src/tools/quality.cpp src/win/version.rc)
  target_link_libraries(OmniSoundProbe PRIVATE omni_win wintrust crypt32)
  add_executable(OmniSoundSetup WIN32 installer/setup.cpp installer/wizard.cpp installer/setup.rc)
  set_property(SOURCE installer/setup.rc APPEND PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/installer/Install.ps1")
  target_link_libraries(OmniSoundSetup PRIVATE omni_win crypt32 shell32 comctl32 uxtheme windowscodecs)
  target_link_options(OmniSoundSetup PRIVATE /MANIFEST:NO)
  install(TARGETS OmniSoundEngine OmniSoundService OmniSoundProbe OmniSoundSetup RUNTIME DESTINATION .)
endif()
if(BUILD_TESTING)
  add_executable(omni_tests tests/core_tests.cpp)
  target_link_libraries(omni_tests PRIVATE omni_core)
  add_test(NAME core COMMAND omni_tests)
  if(WIN32)
    add_executable(omni_win_tests tests/win_tests.cpp)
    target_link_libraries(omni_win_tests PRIVATE omni_win)
    add_test(NAME windows_contracts COMMAND omni_win_tests)
  endif()
endif()
